cmake_minimum_required(VERSION 3.18)
project(rcj_soccer_sim LANGUAGES CXX)

option(RCJ_SIM_GUI "Build the raylib window" ON)
set(RCJ_SIM_COEFFICIENTS_FILE "" CACHE FILEPATH "Strategy coefficients for this simulator build")
add_library(rcj_world world.cpp replay.cpp)
target_compile_features(rcj_world PUBLIC cxx_std_17)
target_include_directories(rcj_world PUBLIC ../main)
target_compile_options(rcj_world PRIVATE -Wall -Wextra -Wpedantic)
if(RCJ_SIM_COEFFICIENTS_FILE)
    if(NOT EXISTS "${RCJ_SIM_COEFFICIENTS_FILE}" OR IS_DIRECTORY "${RCJ_SIM_COEFFICIENTS_FILE}")
        message(FATAL_ERROR "Strategy coefficients file not found: ${RCJ_SIM_COEFFICIENTS_FILE}")
    endif()
    target_compile_definitions(rcj_world PUBLIC
        SOCCER_COEFFICIENTS_FILE="${RCJ_SIM_COEFFICIENTS_FILE}")
endif()

add_executable(rcj-sim main.cpp)
target_link_libraries(rcj-sim PRIVATE rcj_world)
if(RCJ_SIM_GUI)
    set(RCJ_RAYLIB_ROOT "" CACHE PATH "Unpacked raylib 5.5 SDK")
    if(RCJ_RAYLIB_ROOT)
        find_path(RCJ_RAYLIB_INCLUDE raylib.h PATHS "${RCJ_RAYLIB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
        find_library(RCJ_RAYLIB_LIBRARY raylib PATHS "${RCJ_RAYLIB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
        add_library(raylib UNKNOWN IMPORTED)
        set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RCJ_RAYLIB_LIBRARY}"
                              INTERFACE_INCLUDE_DIRECTORIES "${RCJ_RAYLIB_INCLUDE}")
    else()
        find_package(raylib 5.5 QUIET)
    endif()
    if(NOT TARGET raylib)
        include(FetchContent)
        if(POLICY CMP0135)
            cmake_policy(SET CMP0135 NEW)
        endif()
        set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
        FetchContent_Declare(raylib
            URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5)
        FetchContent_MakeAvailable(raylib)
    endif()
    target_sources(rcj-sim PRIVATE window.cpp)
    target_compile_definitions(rcj-sim PRIVATE RCJ_SIM_GUI)
    target_link_libraries(rcj-sim PRIVATE raylib)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(rcj-sim-tests tests.cpp)
    target_link_libraries(rcj-sim-tests PRIVATE rcj_world)
    add_test(NAME sandbox COMMAND rcj-sim-tests)
endif()
